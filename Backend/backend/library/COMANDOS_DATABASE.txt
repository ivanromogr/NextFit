//// ID LEGIBLE /////////
SELECT
  LOWER(CONCAT(
    SUBSTR(HEX(id), 1, 8), '-',
    SUBSTR(HEX(id), 9, 4), '-',
    SUBSTR(HEX(id), 13, 4), '-',
    SUBSTR(HEX(id), 17, 4), '-',
    SUBSTR(HEX(id), 21)
  )) AS id,
  nombre,
  apellido,
  email,
  contrasena,
  rol
FROM usuario;
